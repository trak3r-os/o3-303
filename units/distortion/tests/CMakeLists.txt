# Host tests of the DISTORTION unit (doctest, o3sdk_test_suite): one ctest per *_test.cpp. The oracle test builds the pinned Replicant 303
# DistortionUnit (third_party/replicant303/DSP) on the host and runs the same signals through the original and the port.
file(GLOB test_sources CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*_test.cpp)

# The original, as it is (it is not ours: no strict warnings, doubles allowed). Host only.
set(REPLICANT_DSP "${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/replicant303/DSP")
add_library(distortion_replicant_ref STATIC ${REPLICANT_DSP}/DistortionUnit.cpp)
target_include_directories(distortion_replicant_ref PUBLIC ${REPLICANT_DSP})

# The reference for FUZZ. The port bounds FUZZ's negative branch at -3 (src/distortion.cpp, tanh_pade_floor): the original's fastTanh is a
# Pade approximant that saturates only up to |x| = 3 and climbs again beyond it, which the original's GermaniumFuzz reaches on any hot
# signal. The oracle of FUZZ is therefore the pinned original with that one change, made here by text replacement (the configure step stops
# if the line is not found, i.e. if the pinned file changes) and renamed (DistortionUnitBounded) so that both link into one executable.
# The tests also run FUZZ against the unchanged original where the two must agree, and show where they part.
file(READ "${REPLICANT_DSP}/DistortionUnit.cpp" fuzz_ref_cpp)
file(READ "${REPLICANT_DSP}/DistortionUnit.h" fuzz_ref_h)
set(fuzz_ref_line "fastTanh(biased * 2.1f)")
string(FIND "${fuzz_ref_cpp}" "${fuzz_ref_line}" fuzz_ref_at)
if(fuzz_ref_at EQUAL -1)
  message(FATAL_ERROR "third_party/replicant303/DSP/DistortionUnit.cpp no longer has '${fuzz_ref_line}': update the FUZZ reference of the oracle")
endif()
string(REPLACE "${fuzz_ref_line}" "fastTanh(std::max(biased * 2.1f, -3.0f))" fuzz_ref_cpp "${fuzz_ref_cpp}")
string(REPLACE "DistortionUnit" "DistortionUnitBounded" fuzz_ref_cpp "${fuzz_ref_cpp}")
string(REPLACE "DistortionUnit" "DistortionUnitBounded" fuzz_ref_h "${fuzz_ref_h}")
