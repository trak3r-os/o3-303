# o3-303's own units (DSP only, against o3sdk's dsp layer): one directory each with its CMakeLists.txt, which calls o3sdk_unit().
# The tract DSP (o3-sdk/extras/tract), shared with o3-voc03er. Opt-in on the sdk's side: its CMake never adds it, so it is added here, into a
# binary directory of this build, before the units that link it (unit_tract). The library is unit_tract_dsp.
add_subdirectory(${O3SDK_DIR}/extras/tract ${CMAKE_BINARY_DIR}/extras_tract)
file(GLOB _unit_dirs LIST_DIRECTORIES true CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*")
foreach(_dir IN LISTS _unit_dirs)
  if(EXISTS "${_dir}/CMakeLists.txt")
    add_subdirectory(${_dir})
  endif()
endforeach()
